cmake_minimum_required(VERSION 3.27)
project(FnCPNWindowsUI LANGUAGES CXX)

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../internal/version/VERSION" FNCPN_PRODUCT_VERSION)
string(STRIP "${FNCPN_PRODUCT_VERSION}" FNCPN_PRODUCT_VERSION)
string(REPLACE "." "," FNCPN_VERSION_RESOURCE "${FNCPN_PRODUCT_VERSION}")

if(NOT WEBVIEW2_SDK_ROOT)
  set(WEBVIEW2_SDK_ROOT "$ENV{USERPROFILE}/.nuget/packages/microsoft.web.webview2/1.0.4258.31")
endif()
if(NOT EXISTS "${WEBVIEW2_SDK_ROOT}/build/native/include/WebView2.h")
  message(FATAL_ERROR "WebView2 native SDK is missing. Restore platform/windows/WebView2SDK.csproj or set WEBVIEW2_SDK_ROOT.")
endif()
if(CMAKE_GENERATOR_PLATFORM STREQUAL "ARM64")
  set(WEBVIEW2_ARCH arm64)
else()
  set(WEBVIEW2_ARCH x64)
endif()
set(WEBVIEW2_LOADER_LIB "${WEBVIEW2_SDK_ROOT}/build/native/${WEBVIEW2_ARCH}/WebView2LoaderStatic.lib")
if(NOT EXISTS "${WEBVIEW2_LOADER_LIB}")
  message(FATAL_ERROR "WebView2 loader library is missing for ${WEBVIEW2_ARCH}: ${WEBVIEW2_LOADER_LIB}")
endif()

add_executable(FnCPN WIN32 main.cpp)
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/FnCPN.rc.in" "${CMAKE_CURRENT_BINARY_DIR}/FnCPN.rc" @ONLY)
target_sources(FnCPN PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/FnCPN.rc")
target_compile_features(FnCPN PRIVATE cxx_std_20)
target_compile_definitions(FnCPN PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX "FNCPN_PRODUCT_VERSION=L\"${FNCPN_PRODUCT_VERSION}\"")
target_include_directories(FnCPN PRIVATE "${WEBVIEW2_SDK_ROOT}/build/native/include")
target_link_libraries(FnCPN PRIVATE uxtheme comctl32 shell32 windowsapp ole32 advapi32 winhttp shlwapi "${WEBVIEW2_LOADER_LIB}")
set_target_properties(FnCPN PROPERTIES OUTPUT_NAME "FnCPN.UI")
if(MSVC)
  set_property(TARGET FnCPN PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(FnCPN PRIVATE /W4 /permissive- /utf-8 /EHsc)
  target_link_options(FnCPN PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest")
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(FnCPNNativeTests tests.cpp "${CMAKE_CURRENT_BINARY_DIR}/FnCPN.rc")
  target_compile_features(FnCPNNativeTests PRIVATE cxx_std_20)
  target_compile_definitions(FnCPNNativeTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX "FNCPN_PRODUCT_VERSION=L\"${FNCPN_PRODUCT_VERSION}\"")
  target_include_directories(FnCPNNativeTests PRIVATE "${WEBVIEW2_SDK_ROOT}/build/native/include")
  target_link_libraries(FnCPNNativeTests PRIVATE uxtheme comctl32 shell32 windowsapp ole32 advapi32 winhttp shlwapi "${WEBVIEW2_LOADER_LIB}")
  set_property(TARGET FnCPNNativeTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(FnCPNNativeTests PRIVATE /W4 /permissive- /utf-8 /EHsc)
  target_link_options(FnCPNNativeTests PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest")
  add_test(NAME native-ui-regressions COMMAND FnCPNNativeTests)
  set_tests_properties(native-ui-regressions PROPERTIES TIMEOUT 30)
endif()
